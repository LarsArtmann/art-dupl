```go
// Added to internal/utils/file.go
func FindProjectRoot(startPath string, markers []string) (string, error) {
    absPath, err := filepath.Abs(startPath)
    if err != nil {
        return "", errors.WrapFile(err, startPath, "getting absolute path")
    }

    current := absPath
    maxDepth := 10 // Prevent infinite loops

    for i := 0; i < maxDepth; i++ {
        for _, marker := range markers {
            markerPath := filepath.Join(current, marker)
            if _, err := os.Stat(markerPath); err == nil {
                return current, nil
            }
        }

        parent := filepath.Dir(current)
        if parent == current || parent == "" {
            break
        }
        current = parent
    }

    return "", errors.NewFileError(startPath, "project root not found", nil)
}
```

```go
func FindSQLCConfigs(paths []string) (map[string]string, error) {
    configs := make(map[string]string)

    for _, path := range paths {
        // Search in provided path (existing behavior)
        err := filepath.Walk(path, func(filePath string, info os.FileInfo, err error) error {
            // ... walk logic
            if filename == "sqlc.yaml" || filename == "sqlc.yml" {
                configs[filePath] = filepath.Dir(filePath)
            }
            return nil
        })

        // NEW: Search parent directories for sqlc config
        parentPath, err := utils.FindProjectRoot(path, []string{"sqlc.yaml", "sqlc.yml"})
        if err == nil && parentPath != "" {
            configPath := filepath.Join(parentPath, "sqlc.yaml")
            if _, err := os.Stat(configPath); err == nil {
                configs[configPath] = parentPath
            }
        }
    }

    return configs, nil
}
```

```go
func SafeFileOperations(baseDir string) error {
    root, err := os.OpenRoot(baseDir)
    if err != nil {
        return err
```
